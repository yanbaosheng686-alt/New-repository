<script>
const display = document.getElementById('display');
const errBox = document.getElementById('err');

function append(val){
    errBox.innerText = "";
    display.value += val;
}
function clearAll(){
    display.value = "";
    errBox.innerText = "";
}
function del(){
    display.value = display.value.slice(0,-1);
    errBox.innerText = "";
}
function calcResult(){
    try{
        let res = eval(display.value);
        display.value = res;
        errBox.innerText = "성공 / 成功";
    }catch(e){
        errBox.innerText = "계산 오류 / 计算错误";
    }
}
